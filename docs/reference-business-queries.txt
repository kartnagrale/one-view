WITH trade_summary AS (
    SELECT
        order_id,
--        SUM(last_fill_qty) as total_trade_qty,
        CASE WHEN COUNT(CASE WHEN trd_is_approved = 'T' THEN 1 END) > 0
             THEN 'Approved'
             ELSE 'Rejected'
        END as approval_status
    FROM trade_mstr
    GROUP BY order_id
)
SELECT
    row_number() OVER (
        PARTITION BY FO.ord_auction_ord_no
        ORDER BY
           CASE
                FO.ord_status
                WHEN 'FILL' THEN 1
                ELSE 2
            END,
            FO.ORD_ORD_PRICE ASC,
            FO.ORD_ORD_QTY DESC,
            FO.ord_ord_exch_dt ASC
    ) AS "Rank",
    coalesce (FO.ord_unique_number,'NA') AS "Auction Id",
    FO.ord_auction_ord_no as "Auction Order No.",--
    FO.ord_ord_intrnl_dt AS "Auction Date",
    FO.ord_market_code AS "Market Code",
    FO.ord_contract_code as "Contract code",
    I."Initiator ID",
    I."Initiator Name",
    FO.ord_location AS "Location",
    FO.ord_tm_id AS "CMID",
    RM.org_nm AS "Bidder Name",
    RM.ro_cty as "Bidder City",
    FO.ord_ord_price AS "Lowest Price",
    FO.ord_ord_qty AS "Quantity",
    FO.ord_auction_short_code AS "Product Name",
    coalesce(FO.ord_paramvalue11,'') AS "Product Description",
    FO.ord_remarks AS "Auction Description",
    FO.ord_status AS "Order Status",
    COUNT(*) OVER (PARTITION BY FO.ord_auction_ord_no) AS "Bid Count",
    coalesce(EM.error,'') AS "Rejection Reason",
    coalesce(ts.approval_status,'Rejected') as "Auction Status"
--    case when tm.trd_is_approved='T' then 'Approved' else 'Rejected' end as "Auction status"
FROM (
    SELECT
        ord_contract_code, ord_auction_ord_no, ord_unique_number, ord_ord_intrnl_dt,
        ord_market_code, ord_location, ord_tm_id, ord_ord_price, ord_ord_qty,
        ord_auction_short_code, ord_paramvalue11, ord_remarks, ord_status,
        ord_rej_reason, ord_ord_exch_dt, ord_ord_type,ord_exch_ord_no,ord_ord_side
    FROM fo_orders  
    UNION ALL
    SELECT
        ord_contract_code, ord_auction_ord_no, ord_unique_number, ord_ord_intrnl_dt,
        ord_market_code, ord_location, ord_tm_id, ord_ord_price, ord_ord_qty,
        ord_auction_short_code, ord_paramvalue11, ord_remarks, ord_status,
        ord_rej_reason, ord_ord_exch_dt, ord_ord_type,ord_exch_ord_no,ord_ord_side
    FROM fo_orders_purge_01feb2024 FOP
    where NOT EXISTS (
        SELECT 1 FROM fo_orders FO
        WHERE FO.ord_unique_number = FOP.ord_unique_number
    )
) FO
INNER JOIN registration_mstr RM ON FO.ord_tm_id = RM.org_id and rm.is_mock = 'F'
LEFT JOIN errorcodes_mstr EM ON FO.ord_rej_reason = EM.errorcode
left join trade_summary ts on FO.ord_exch_ord_no = ts.order_id
left join trade_mstr tm on fo.ord_exch_ord_no = tm.order_id and tm.trd_is_approved='T'
LEFT JOIN (select
    distinct
        fo.ord_tm_id as "Initiator ID",
    um.tm_name as "Initiator Name",
    fo.ord_exch_ord_no
from(select
ord_tm_id,
        ord_exch_ord_no
    from
        fo_orders
    where
        ord_ord_type = 'AO'
union all
    select
        ord_tm_id,
        ord_exch_ord_no
    from
        fo_orders_purge_01feb2024
    where
        ord_ord_type = 'AO'
    ) fo
inner join tm_mstr um on
    fo.ord_tm_id = um.tm_id) I on I.ord_exch_ord_no = FO.ord_auction_ord_no
WHERE
    FO.ord_ord_intrnl_dt::date BETWEEN '2022/01/01'::DATE AND '2025/10/31'::DATE
    AND FO.ord_ord_type = 'AB'
    and FO.ord_ord_side='S'
    and FO.ord_contract_code in (select contract_code from temp_contracts_03_11_chemicals);

select
	fo.ord_status as "Order Status",
	fo.ord_ord_exch_dt as "Order date",
	fo.ord_market_type as "Market Code",
	fo.ord_contract_code as "Contract Code",
	fo.ord_market_code as "Market Code",
	fo.ord_comm_code as "Commodity Code",
	fo.ord_dealer_id as "Dealer Id",
	fo.ord_ord_side as "Side",
	fo.ord_internal_ord_no as "Internal Order no.",
	fo.ord_exch_ord_no as "Exchange Order no.",
	fo.ord_tm_id as "TM ID",
	fo.ord_trd_acc_no as "TAC ID",
	fo.ord_ord_price as "Price",
	fo.ord_ord_qty as "Qty",
	fo.ord_price_unit as "Price Unit",
	fo.ord_deliverymode as "Delivery Mode",
	fo.ord_location as "Location",
	fo.ord_pricingdate as "Pricing Date",
	fo.ord_qty_unit as "Qty Unit",
	fo.ord_unique_number as "Unique No",
	fo.ord_pending_qty as "Pending Qty",
	fo.ord_ord_value as "Order Value",
	fo.ord_ord_type as "Order Type",
	fo.global_client as "Global Client",
	em.error as "Rejection Error",
	fo.ord_remarks as "Remarks",
	fo.ord_auction_short_code as "Auction Short Code",
	ab.rk as Bid_rank,
	fo.ord_bidding_start_time as "Bidding Start Time",
	fo.ord_bidding_end_time as "Bidding End Time",
	fo.ord_sta as "STA %",
	fo.ord_match_qty as "MMQ",
	fo.ord_min_bid_qty as "MBQ",
	fo.ord_paramvalue4,
	fo.ord_paramvalue5,
	fo.ord_paramvalue6,
	fo.ord_paramvalue7,
	fo.ord_paramvalue8,
	fo.ord_paramvalue9,
	fo.ord_paramvalue10,
	fo.ord_paramvalue11,
	fo.ord_paramvalue12,
	fo.ord_paramvalue13
from
	fo_orders fo
left join (
	select
		ord_internal_ord_no,
		case
			when ord_ord_side = 'B' then 'H'
			when ord_ord_side = 'S' then 'L'
		end || rank() over (
            partition by ord_auction_ord_no
	order by
		case
			when fo.ord_status in ('FILL', 'PFILL') then 1
			when fo.ord_status in ('KNKOFF') then 1
			else 2
		end asc,
		case
			when fo.ord_ord_side = 'B' then fo.ord_ord_price
		end desc,
		case
			when fo.ord_ord_side = 'S' then fo.ord_ord_price
		end asc,
		fo.ord_ord_qty desc,
		fo.ord_ord_exch_dt asc
        ) as rk
	from
		fo_orders fo
	inner join time_session_map tsm on
		(
        fo.ord_token_id = tsm.token_id
			and tsm.session_id = 2
			and tsm.token_is_loaded = 'C'
    )
	where
		ord_ord_type = 'AB'
		and ORD_MARKET_TYPE = 'EM'
		and ORD_STATUS not in ('REJ', 'REPLACE')
		and ORD_COMM_CODE like '%SCRAP%'
		and DATE(ORD_ORD_INTRNL_DT) between DATE('2025-01-01') and DATE('2025-12-31')
) ab on
	fo.ord_internal_ord_no = ab.ord_internal_ord_no
left join PROVISIONAL_TRADE_MSTR ptm 
    on
	fo.ORD_EXCH_ORD_NO = ptm.ORDER_ID
left join ERRORCODES_MSTR em 
    on
	em.ERRORCODE = fo.ORD_REJ_REASON
where
	1 = 1
	and fo.ORD_MARKET_TYPE = 'EM'
	and fo.ORD_STATUS not in ('UPLOAD', 'INVALID')
	and fo.ORD_COMM_CODE like '%SCRAP%'
	and DATE(fo.ORD_ORD_INTRNL_DT) between DATE('2025-01-01') and DATE('2025-12-31')
order by
	fo.ORD_ORD_EXCH_DT desc;

select

agg.side as "Side",
agg.total_fill_value as "Transaction Value",
agg.total_fill_qty as "Transaction Volume",
agg.future_symbol as "Commodity",
agg.symbol as "Contract",
coalesce(agg.total_cost,0) as "Transaction Charge"
from
registration_mstr rm
inner join user_mstr um on rm.org_id=um.org_id
inner join (select
tm.tm_id,
to_char(tm.trade_datetime, 'YYYY-MM') as trade_datemonth,
tm.side,
sum(last_fill_value) as total_fill_value ,
sum(last_fill_qty) as total_fill_qty ,
tm.future_symbol ,
tm.symbol,
sum(db.amount) as total_cost
from
trade_mstr tm
left join mktpgprd.ct_unified_cstracker cs on
tm.execution_id = cs.trade_no
left join (
select
cm_id ,
obligation_no,
sum(amount) as amount
from
mktpgprd.ct_dues_book
where
charge_id in (36,
37)
group by
cm_id,
obligation_no
) db on
db.obligation_no = cs.intention_no
where
tm.trd_is_approved = 'T'
group by
tm.tm_id,
to_char(tm.trade_datetime, 'YYYY-MM'),
tm.side,
tm.future_symbol,
tm.symbol) agg on
agg.tm_id = rm.org_id
where

WITH daily_bids AS (
    -- 1. Count of bids from 11:00 AM to 11:45 AM per day
    SELECT
        ord_created_dt::date AS auction_date,
        COUNT(*) AS max_bids_11_to_1145
    FROM fo_orders
    WHERE ord_ord_type = 'AB'
    and global_client ='NAFED'
      AND ord_created_dt::date >= '2025-01-01'
      AND ord_created_dt::time >= '11:00:00'
      AND ord_created_dt::time <= '11:45:00'
    GROUP BY ord_created_dt::date
),
last_5_min_bids AS (
    -- 2. Count of bids in the last 5 minutes (11:40 AM to 11:45 AM) per day
    SELECT
        ord_created_dt::date AS auction_date,
        COUNT(*) AS max_bids_last_5_min
    FROM fo_orders
    WHERE ord_ord_type = 'AB'
    and global_client ='NAFED'
      AND ord_created_dt::date >= '2025-01-01'
      AND ord_created_dt::time >= '11:40:00'
      AND ord_created_dt::time <= '11:45:00'
    GROUP BY ord_created_dt::date
),
concurrent_bids AS (
    -- 3. Max concurrent bids (bids placed in the exact same second) per day
    SELECT
        auction_date,
        MAX(bids_per_second) AS max_concurrent_bids
    FROM (
        SELECT
            ord_created_dt::date AS auction_date,
            date_trunc('second', ord_created_dt) AS bid_second,
            COUNT(*) AS bids_per_second
        FROM fo_orders
        WHERE ord_ord_type = 'AB'
        and global_client ='NAFED'
          AND ord_created_dt::date >= '2025-01-01'
        GROUP BY ord_created_dt::date, date_trunc('second', ord_created_dt)
    ) sub
    GROUP BY auction_date
)
SELECT
    ROW_NUMBER() OVER (ORDER BY d.auction_date ASC) AS "Sr No",
    d.auction_date AS "Date of auction",
    d.max_bids_11_to_1145 AS "Maximum count of bids received in NAFED from 11 am to 11.45 am",
    COALESCE(l5.max_bids_last_5_min, 0) AS "Maximum count of bids received in the last 5 minutes",
    COALESCE(cb.max_concurrent_bids, 0) AS "Count of Maximum concurrent bids received(bids/sec)"
FROM daily_bids d
LEFT JOIN last_5_min_bids l5
    ON d.auction_date = l5.auction_date
LEFT JOIN concurrent_bids cb
    ON d.auction_date = cb.auction_date
ORDER BY d.auction_date ASC;

    UNION ALL
    SELECT order_id, last_fill_qty, last_fill_value, trd_is_approved FROM trade_mstr_bkp_20260723

BaseData AS (
    SELECT
        CASE
            WHEN EXTRACT(MONTH FROM fo.ord_bidding_start_time) <= 3
                THEN (EXTRACT(YEAR FROM fo.ord_bidding_start_time) - 1) || '-' || TO_CHAR(fo.ord_bidding_start_time, 'YY')
            ELSE EXTRACT(YEAR FROM fo.ord_bidding_start_time) || '-' || TO_CHAR(fo.ord_bidding_start_time + INTERVAL '1 year', 'YY')
